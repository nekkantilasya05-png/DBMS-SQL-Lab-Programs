USE UniversityDB;

-- ==========================================
-- SET 4
-- a) Employee having highest salary
-- b) Employee View
-- c) Students enrolled in specific course
-- d) GRANT and REVOKE
-- ==========================================

-- a) HIGHEST SALARY USING NESTED QUERY

SELECT *
FROM Employee
WHERE salary = (
    SELECT MAX(salary)
    FROM Employee
);

-- b) CREATE VIEW AND RETRIEVE RECORDS

CREATE OR REPLACE VIEW Employee_Details AS
SELECT emp_id, emp_name, department, salary, city
FROM Employee;

SELECT * FROM Employee_Details;

-- c) STUDENTS ENROLLED IN DBMS COURSE

SELECT
    s.student_id,
    s.student_name,
    c.course_name,
    e.grade
FROM Student s
JOIN Enrollment e
ON s.student_id = e.student_id
JOIN Course c
ON e.course_id = c.course_id
WHERE c.course_name = 'DBMS';

-- d) DCL COMMANDS

-- Run as administrator:
-- CREATE USER 'labuser'@'localhost' IDENTIFIED BY 'Lab@123';

GRANT SELECT
ON UniversityDB.Employee
TO 'labuser'@'localhost';

REVOKE SELECT
ON UniversityDB.Employee
FROM 'labuser'@'localhost';
