USE UniversityDB;

-- ==========================================
-- SET 8
-- a) Employees earning less than David
-- b) View using suitable tables
-- c) Count students enrolled in each course
-- d) GRANT and REVOKE
-- ==========================================

-- a) EMPLOYEES EARNING LESS THAN DAVID

SELECT *
FROM Employee
WHERE salary < (
    SELECT salary
    FROM Employee
    WHERE emp_name = 'David'
);

-- b) VIEW USING MULTIPLE TABLES

CREATE OR REPLACE VIEW Student_Course_View AS
SELECT
    s.student_id,
    s.student_name,
    c.course_name,
    e.grade
FROM Student s
JOIN Enrollment e
ON s.student_id = e.student_id
JOIN Course c
ON e.course_id = c.course_id;

SELECT * FROM Student_Course_View;

-- c) NUMBER OF STUDENTS ENROLLED IN EACH COURSE

SELECT
    c.course_id,
    c.course_name,
    COUNT(e.student_id) AS Student_Count
FROM Course c
LEFT JOIN Enrollment e
ON c.course_id = e.course_id
GROUP BY c.course_id, c.course_name;

-- d) GRANT

GRANT SELECT
ON UniversityDB.Student_Course_View
TO 'labuser'@'localhost';

-- REVOKE

REVOKE SELECT
ON UniversityDB.Student_Course_View
FROM 'labuser'@'localhost';
