-- a) STUDENTS BELONGING TO AIML DEPARTMENT

SELECT s.student_id, s.student_name
FROM Student s
JOIN Department d
ON s.dept_id = d.dept_id
WHERE d.dept_name = 'AIML';

-- c) AGGREGATE FUNCTIONS

SELECT
    department,
    SUM(salary) AS Total_Salary,
    AVG(salary) AS Average_Salary,
    MIN(salary) AS Minimum_Salary,
    MAX(salary) AS Maximum_Salary,
    COUNT(*) AS Employee_Count
FROM Employee
GROUP BY department
HAVING AVG(salary) > 50000;

-- d) SET OPERATIONS

-- UNION
SELECT city FROM Employee
WHERE department = 'IT'
UNION
SELECT city FROM Employee
WHERE department = 'Finance';

-- INTERSECTION
SELECT city FROM Employee
WHERE department = 'IT'
INTERSECT
SELECT city FROM Employee
WHERE department = 'Finance';

-- DIFFERENCE
-- MySQL alternative for EXCEPT/MINUS
SELECT city
FROM Employee
WHERE department = 'IT'
AND city NOT IN (
    SELECT city
    FROM Employee
    WHERE department = 'Finance'
);
