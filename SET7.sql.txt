USE UniversityDB;

-- ==========================================
-- SET 7
-- a) Student enrollments with grades
-- b) Change city using UPDATE
-- c) Numeric and Date/Time functions
-- d) Joins between Department and Employee
-- ==========================================

-- a) STUDENT ENROLLMENTS WITH GRADES

SELECT
    s.student_name,
    c.course_name,
    e.grade
FROM Student s
JOIN Enrollment e
ON s.student_id = e.student_id
JOIN Course c
ON e.course_id = c.course_id;

-- b) CHANGE CITY OF SPECIFIED EMPLOYEE

UPDATE Employee
SET city = 'Hyderabad'
WHERE emp_id = 101;

SELECT * FROM Employee
WHERE emp_id = 101;

-- c) NUMERIC FUNCTIONS

SELECT
    emp_name,
    salary,
    ROUND(salary / 12, 2) AS Monthly_Salary,
    CEIL(salary / 1000) AS Salary_Ceil,
    FLOOR(salary / 1000) AS Salary_Floor,
    ABS(salary - 60000) AS Salary_Difference
FROM Employee;

-- DATE/TIME FUNCTIONS

SELECT
    emp_name,
    joining_date,
    YEAR(joining_date) AS Joining_Year,
    MONTH(joining_date) AS Joining_Month,
    DAY(joining_date) AS Joining_Day,
    CURDATE() AS Current_Date
FROM Employee;

-- d) JOIN BETWEEN DEPARTMENT AND EMPLOYEE

SELECT
    d.dept_id,
    d.dept_name,
    e.emp_id,
    e.emp_name,
    e.salary
FROM Department d
INNER JOIN Employee e
ON d.dept_name = e.department;
